\begin{frame}[fragile]\frametitle{Field Experience: Marks and Format}
\begin{itemize}
\item \textbf{Marks}: 10 of the 140 practical marks.
\item \textbf{Where it is assessed}: inside the practical assessment, which lasts at most one hour per candidate.
\item \textbf{What it asks}: show that you have taught real people, for example classes in a park, a society or a workplace.
\item \textbf{Preparation}: keep a short log of the classes you have taken (place, group, what you taught, feedback) and be ready to talk about one of them.
\end{itemize}
{\tiny (Ref: YCB, Guidelines for Yoga Professionals, Version 0.2; the log is a study suggestion, not an official requirement)}
\end{frame}
